:: ENCODE.BAT -- Command prompt shell for ENCODE.EXE [CLD rev.9/19/15]
@echo off
if %1! == ! goto :error1
for %%Z in (%0) do set FYL_STEM=%%~dpnZ
set FYL_MODE=
if /i %1 == b (
  set FYL_MODE=B
  shift 
  if %1! == ! goto :error1 )
if not exist %1 goto error2
